## 快速测试

### 方式 1: 运行内置示例

```rust
// 在 src-tauri/src/ 目录下创建 test_agent_workflow.rs

use ifai::agent_system::workflow::*;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    println!("=== 多智能体协作系统测试 ===\n");

    // 1. 创建简单工作流
    let mut workflow = Workflow::new("test-workflow", "测试工作流")
        .with_description("探索和审查代码");

    workflow
        .add_node(WorkflowNode::new("explore", AgentType::Explore)
            .with_label("探索代码结构"))
        .add_node(WorkflowNode::new("review", AgentType::Review)
            .with_label("代码质量审查"))
        .add_edge(WorkflowEdge::new("explore", "review"));

    // 2. 验证工作流
    println!("📝 验证工作流...");
    workflow.validate()?;
    println!("✅ 工作流验证成功\n");

    // 3. 调度工作流
    println!("📋 调度工作流...");
    let schedule = WorkflowScheduler::schedule(&workflow)?;
    println!("✅ 调度成功");
    println!("   执行顺序: {:?}", schedule.execution_order);
    println!();

    // 4. 执行工作流
    println!("🚀 执行工作流...");
    let runner = WorkflowRunner::with_default_config(workflow)?;
    let result = runner.run().await?;

    // 5. 显示结果
    println!("✅ 执行完成");
    println!("   状态: {:?}", result.status);
    println!("   开始时间: {:?}", result.started_at);
    println!("   结束时间: {:?}", result.completed_at);
    println!();

    // 6. 显示节点结果
    for (node_id, node_result) in &result.node_results {
        println!("📊 节点 {}: {:?}", node_id, node_result.status);
    }

    Ok(())
}
```

**运行方式**:
```bash
cd src-tauri
cargo run --bin test_agent_workflow
```

```yaml

  - from: "explore"
    to: "review"
  - from: "review"
    to: "refactor"
```

**运行测试**:
```rust
use ifai::agent_system::workflow::*;

    println!("   边数: {}", workflow.edges.len());

    // 执行工作流
    let runner = WorkflowRunner::with_default_config(workflow)?;
```

### 3. 作为独立服务运行

```rust
// src-tauri/src/workflow_service.rs

use std::sync::Arc;
use tokio::sync::RwLock;

pub struct WorkflowService {
    // 存储运行中的工作流
    running_workflows: Arc<RwLock<HashMap<String, WorkflowRunner>>>,
}

impl WorkflowService {
    pub fn new() -> Self {
        Self {
            running_workflows: Arc::new(RwLock::new(HashMap::new())),
        }
    }

    pub async fn submit_workflow(
        &self,
        workflow: Workflow,
    ) -> Result<String, String> {
        let workflow_id = workflow.id.clone();

        // 创建运行器
        let runner = WorkflowRunner::with_default_config(workflow)?;

        // 存储运行器
        let mut workflows = self.running_workflows.write().await;
        workflows.insert(workflow_id.clone(), runner);

        Ok(workflow_id)
    }

    pub async fn get_workflow_status(
        &self,
        workflow_id: &str,
    ) -> Result<WorkflowStatus, String> {
        let workflows = self.running_workflows.read().await;
        let runner = workflows.get(workflow_id)
            .ok_or("工作流不存在")?;

        Ok(runner.get_status().await)
    }

    pub async fn cancel_workflow(
        &self,
        workflow_id: &str,
    ) -> Result<(), String> {
        let mut workflows = self.running_workflows.write().await;
        let runner = workflows.get_mut(workflow_id)
            .ok_or("工作流不存在")?;

        runner.cancel().await?;
        workflows.remove(workflow_id);

        Ok(())
    }
}
```

## 生产部署

### 1. 配置优化

```rust
use ifai::agent_system::workflow::*;

// 生产环境推荐配置
pub fn production_config() -> RunnerConfig {
    RunnerConfig {
        // 根据 CPU 核心数设置并发数
        max_concurrent_nodes: num_cpus::get(),

        // 设置合理的超时时间（10分钟）
        node_timeout_secs: 600,

        // 失败重试 3 次
        max_retries: 3,

        // 任何节点失败立即停止
        fail_fast: true,
    }
}

// 开发环境配置
pub fn development_config() -> RunnerConfig {
    RunnerConfig {
        max_concurrent_nodes: 2,  // 限制并发以便调试
        node_timeout_secs: 300,   // 5 分钟超时
        max_retries: 1,           // 只重试一次
        fail_fast: false,         // 不立即失败，查看所有结果
    }
}
```

### 2. 错误处理

```rust
#[tauri::command]
pub async fn execute_workflow_safe(
    workflow: Workflow,
) -> Result<serde_json::Value, String> {
    // 1. 验证
    workflow.validate()
        .map_err(|e| format!("验证失败: {:?}", e))?;

    // 2. 调度
    let schedule = WorkflowScheduler::schedule(&workflow)
        .map_err(|e| format!("调度失败: {:?}", e))?;

    log::info!("执行计划: {:?}", schedule.execution_order);

    // 3. 执行
    let runner = WorkflowRunner::with_default_config(workflow)
        .map_err(|e| e.to_string())?;

    let result = runner.run().await;

    // 4. 处理结果
    match result {
        Ok(result) => {
            log::info!("工作流完成: {:?}", result.status);

            // 检查是否有失败的节点
            if !result.is_all_success() {
                let failed: Vec<_> = result.node_results
                    .iter()
                    .filter(|(_, r)| r.status != NodeStatus::Completed)
                    .collect();

                log::warn!("失败的节点: {:?}", failed);
            }

            Ok(serde_json::to_value(result).unwrap())
        }
        Err(e) => {
            log::error!("工作流执行失败: {:?}", e);
            Err(format!("执行失败: {}", e))
        }
    }
}
```

### 3. 监控和日志

```rust
use tracing::{info, warn, error};

pub struct MonitoredWorkflowRunner {
    runner: WorkflowRunner,
}

impl MonitoredWorkflowRunner {
    pub async fn run_with_monitoring(&self) -> Result<WorkflowResult> {
        let start = std::time::Instant::now();

        info!("开始执行工作流");

        match self.runner.run().await {
            Ok(result) => {
                let duration = start.elapsed();

                info!("工作流执行完成");
                info!("  状态: {:?}", result.status);
                info!("  耗时: {:?}", duration);
                info!("  节点数: {}", result.node_results.len());

                // 统计成功/失败
                let success = result.node_results
                    .values()
                    .filter(|r| r.status == NodeStatus::Completed)
                    .count();

                let failed = result.node_results.len() - success;

                info!("  成功: {}, 失败: {}", success, failed);

                Ok(result)
            }
            Err(e) => {
                error!("工作流执行失败: {:?}", e);
                Err(e)
            }
        }
    }
}
```

## 常见问题

### Q1: 如何调试工作流？

**A**: 启用详细日志并使用开发配置：

```bash
RUST_LOG=debug RUST_BACKTRACE=1 cargo test agent_system::workflow --lib -- --nocapture
```

### Q2: 如何处理长时间运行的节点？

**A**: 增加超时时间并实现进度报告：

```rust
let config = RunnerConfig {
    node_timeout_secs: 3600,  // 1 小时
    ..Default::default()
};
```

### Q3: 如何重用工作流组件？

**A**: 创建工作流模板：

```rust
pub fn create_code_review_template() -> Workflow {
    let mut workflow = Workflow::new("code-review-template", "代码审查模板");

    workflow
        .add_node(WorkflowNode::new("explore", AgentType::Explore))
        .add_node(WorkflowNode::new("review", AgentType::Review))
        .add_node(WorkflowNode::new("refactor", AgentType::Refactor))
        .add_edge(WorkflowEdge::new("explore", "review"))
        .add_edge(WorkflowEdge::new("review", "refactor"));

    workflow
}

// 使用
let mut workflow = create_code_review_template();
workflow.id = "my-review".to_string();
workflow.name = "我的代码审查".to_string();
```

### Q4: 如何测试工作流性能？

**A**: 使用基准测试：

```rust
use std::time::Instant;

#[tokio::test]
async fn benchmark_workflow_execution() {
    let workflow = create_test_workflow();

    let start = Instant::now();
    let result = WorkflowRunner::with_default_config(workflow)
        .unwrap()
        .run()
        .await
        .unwrap();
    let duration = start.elapsed();

    println!("执行时间: {:?}", duration);
    println!("节点数: {}", result.node_results.len());

    assert!(duration.as_secs() < 10, "执行时间过长");
}
```

### Q5: 如何集成到现有项目？

**A**: 参考以下步骤：

1. **添加依赖**:

   ```toml
   [dependencies]
   ```

3. **定义工作流**:

   ```yaml
   name: "我的工作流"
   ```
